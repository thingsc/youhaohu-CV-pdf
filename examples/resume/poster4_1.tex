\begin{cvposter}{Research Topic: Control Algorithms}
  \vfill
  \postertitle{Adaptive Finite-/Fixed-time Sliding Mode Control with Disturbance Observers for Electronic Vehicle Throttle System}

  \postersubtitle{Published in 
  \textit{IEEE Transactions on Vehicular Technology}, 
  \textit{Neural Computing and Application}, 
  \textit{Mechanical Systems and Signal Processing}}

  \posterabstract{This project studies finite-time and fixed-time sliding mode control for electronic throttle systems with parametric uncertainties, lumped uncertainty, and actuator constraints. The feedback controller adopts fixed-time sliding mode (FTSM) [1,5], recursive terminal sliding mode (RTSM) [2], dynamic sliding mode (DSM) [3], and nonsingular terminal sliding mode (NTSM) [4] controller to ensure the rapid and robust control property. For meeting requirements in all operation conditions, parameters adaptive mechanism and disturbance estimation methods are further developed to tune the parameters online [1-3] and compensate for the lumped uncertainty in real time [1-5].}
